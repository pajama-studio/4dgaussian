67struct ResearchSplat {
68    base: Vec3,
69    temporal_center: f32,
70    temporal_scale: f32,
71    motion_linear: Vec3,
72    motion_quadratic: Vec3,
73    motion_cubic: Vec3,
74    opacity_logit: f32,
75}
76
77impl ResearchSplat {
78    fn sample(self, time: f32) -> (Vec3, f32) {
79        let dt = time - self.temporal_center;
80        let dt2 = dt * dt;
81        let position = self.base
82            + self.motion_linear * dt
83            + self.motion_quadratic * dt2
84            + self.motion_cubic * (dt2 * dt);
85        let temporal = (-(dt / self.temporal_scale).powi(2)).exp();
86        (position, sigmoid(self.opacity_logit) * temporal)
87    }
88}

103struct GpuTimer {
104    query_set: wgpu::QuerySet,
105    resolve_buffer: wgpu::Buffer,
106    readbacks: Vec<TimestampReadback>,
107    next_readback: usize,
108    latest_ms_bits: Arc<AtomicU64>,
109    timestamp_period_ns: f64,
110}
111
112impl GpuTimer {
113    fn new(device: &wgpu::Device, queue: &wgpu::Queue) -> Self {
114        let query_set = device.create_query_set(&wgpu::QuerySetDescriptor {
115            label: Some("stg-render-timestamps"),
116            ty: wgpu::QueryType::Timestamp,
117            count: 2,
118        });
119        let resolve_buffer = device.create_buffer(&wgpu::BufferDescriptor {
120            label: Some("stg-timestamp-resolve"),
121            size: GPU_TIMESTAMP_BYTES,
122            usage: wgpu::BufferUsages::QUERY_RESOLVE | wgpu::BufferUsages::COPY_SRC,
123            mapped_at_creation: false,
124        });
125        let readbacks = (0..GPU_TIMESTAMP_READBACK_SLOTS)
126            .map(|_| TimestampReadback {
127                buffer: device.create_buffer(&wgpu::BufferDescriptor {
128                    label: Some("stg-timestamp-readback"),
129                    size: GPU_TIMESTAMP_BYTES,
130                    usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
131                    mapped_at_creation: false,
132                }),
133                pending: Arc::new(AtomicBool::new(false)),
134            })
135            .collect();
136        Self {
137            query_set,
138            resolve_buffer,
139            readbacks,
140            next_readback: 0,
141            latest_ms_bits: Arc::new(AtomicU64::new((-1.0_f64).to_bits())),
142            timestamp_period_ns: queue.get_timestamp_period() as f64,
143        }
144    }
145
146    fn reserve_readback(&mut self) -> Option<(wgpu::Buffer, Arc<AtomicBool>)> {
147        for _ in 0..self.readbacks.len() {
148            let index = self.next_readback;
149            self.next_readback = (self.next_readback + 1) % self.readbacks.len();
150            let readback = &self.readbacks[index];
151            if !readback.pending.load(AtomicOrdering::Relaxed) {
152                readback.pending.store(true, AtomicOrdering::Relaxed);
153                return Some((readback.buffer.clone(), readback.pending.clone()));
154            }
155        }
156        None
157    }

377    pub fn render(
378        &mut self,
379        time_seconds: f32,
380        yaw: f32,
381        pitch: f32,
382        distance: f32,
383        width: u32,
384        height: u32,
385        calibrated_camera: js_sys::Float32Array,
386    ) -> Result<(), JsValue> {
387        let prepare_started = Instant::now();
388        self.inspection_frame = None;
389        let width = width.max(1);
390        let height = height.max(1);
391        if self.config.width != width || self.config.height != height {
392            self.config.width = width;
393            self.config.height = height;
394            self.surface.configure(&self.device, &self.config);
395        }
396
397        let (view, projection, viewport) = if calibrated_camera.length() >= 12 {
398            let value = |index| calibrated_camera.get_index(index);
399            let eye = Vec3::new(value(0), value(1), value(2));
400            let mut forward = Vec3::new(value(3), value(4), value(5)).normalize_or_zero();
401            let mut up = Vec3::new(value(6), value(7), value(8)).normalize_or_zero();
402            if forward.length_squared() < 0.5 {
403                forward = Vec3::Z;
404            }
405            if up.length_squared() < 0.5 {
406                up = Vec3::NEG_Y;
407            }
408            let capture_aspect = (value(10) / value(11).max(1.0)).clamp(0.5, 3.0);
409            let surface_aspect = width as f32 / height as f32;
410            let viewport = if surface_aspect > capture_aspect {
411                let viewport_width = height as f32 * capture_aspect;
412                [
413                    (width as f32 - viewport_width) * 0.5,
414                    0.0,
415                    viewport_width,
416                    height as f32,
417                ]
418            } else {
419                let viewport_height = width as f32 / capture_aspect;
420                [
421                    0.0,
422                    (height as f32 - viewport_height) * 0.5,
423                    width as f32,
424                    viewport_height,
425                ]
426            };
427            (
428                Mat4::look_at_rh(eye, eye + forward, up),
429                Mat4::perspective_rh(value(9).clamp(0.2, 2.6), capture_aspect, 0.05, 180.0),
430                viewport,
431            )
432        } else {
433            let pitch = pitch.clamp(-0.55, 0.7);
434            let distance = distance.clamp(9.0, 34.0);
435            let orbit = Vec3::new(
436                yaw.sin() * pitch.cos(),
437                -pitch.sin(),
438                -yaw.cos() * pitch.cos(),
439            );
440            let eye = SCENE_TARGET + orbit * distance;
441            (
442                Mat4::look_at_rh(eye, SCENE_TARGET, Vec3::NEG_Y),
443                Mat4::perspective_rh(
444                    69.5_f32.to_radians(),
445                    width as f32 / height as f32,
446                    0.05,
447                    180.0,
448                ),
449                [0.0, 0.0, width as f32, height as f32],
450            )
451        };
452        let view_proj = projection * view;
453        let normalized_time = time_seconds.rem_euclid(LOOP_SECONDS) / LOOP_SECONDS;
454
455        self.frame.clear();
456        for (index, source) in self.source.iter().enumerate() {
457            let (center, opacity) = source.sample(normalized_time);
458            if opacity < 1.0 / 255.0 {
459                continue;
460            }
461            let clip = view_proj * center.extend(1.0);
462            if clip.w <= 0.0 {
463                continue;
464            }
465            let ndc = clip.truncate() / clip.w;
466            if ndc.x.abs() > 1.35 || ndc.y.abs() > 1.35 || ndc.z < 0.0 || ndc.z > 1.0 {
467                continue;
468            }
469            let depth = -(view * center.extend(1.0)).z;
470            self.frame.push((depth, index as u32));
471        }
472        let sort_started = Instant::now();
473        self.frame
474            .sort_unstable_by(|a, b| b.0.partial_cmp(&a.0).unwrap_or(Ordering::Equal));
475        self.telemetry.sort_ms = sort_started.elapsed().as_secs_f64() * 1000.0;
476
477        let camera = CameraUniform {
478            view_proj: view_proj.to_cols_array_2d(),
479            viewport: [
480                viewport[2],
481                viewport[3],
482                1.0 / viewport[2],
483                1.0 / viewport[3],
484            ],
485            scene: [normalized_time, 0.0, 0.0, 0.0],
486        };
487        self.queue
488            .write_buffer(&self.camera_buffer, 0, bytemuck::bytes_of(&camera));
489        self.sorted_indices.clear();
490        self.sorted_indices
491            .extend(self.frame.iter().map(|(_, index)| *index));
492        self.queue.write_buffer(
493            &self.index_buffer,
494            0,
495            bytemuck::cast_slice(&self.sorted_indices),
496        );
497        self.telemetry.visible = self.sorted_indices.len() as u32;
498        self.telemetry.upload_bytes =
499            (self.sorted_indices.len() * std::mem::size_of::<u32>()) as u32;
500        self.telemetry.prepare_ms = prepare_started.elapsed().as_secs_f64() * 1000.0;
501
502        let inspection_frame = inspection::InspectionFrame {
503            time: normalized_time,
504            view,
505            view_proj,
506            viewport,
507            eye: view.inverse().transform_point3(Vec3::ZERO),
508        };
509        let frame = match self.surface.get_current_texture() {
510            wgpu::CurrentSurfaceTexture::Success(frame)
511            | wgpu::CurrentSurfaceTexture::Suboptimal(frame) => frame,
512            wgpu::CurrentSurfaceTexture::Outdated | wgpu::CurrentSurfaceTexture::Lost => {
513                self.surface.configure(&self.device, &self.config);
514                return Ok(());
515            }
516            wgpu::CurrentSurfaceTexture::Timeout | wgpu::CurrentSurfaceTexture::Occluded => {
517                return Ok(())
518            }
519            wgpu::CurrentSurfaceTexture::Validation => {
520                return Err(JsValue::from_str("WebGPU surface validation failed"))
521            }
522        };
523        let view = frame.texture.create_view(&Default::default());
524        let mut encoder = self
525            .device
526            .create_command_encoder(&wgpu::CommandEncoderDescriptor {
527                label: Some("stg-frame"),
528            });
529        {
530            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
531                label: Some("stg-splats"),
532                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
533                    view: &view,
534                    resolve_target: None,
535                    depth_slice: None,
536                    ops: wgpu::Operations {
537                        load: wgpu::LoadOp::Clear(wgpu::Color {
538                            r: 0.0,
539                            g: 0.0,
540                            b: 0.0,
541                            a: 1.0,
542                        }),
543                        store: wgpu::StoreOp::Store,
544                    },
545                })],
546                depth_stencil_attachment: None,
547                timestamp_writes: self.gpu_timer.as_ref().map(|timer| {
548                    wgpu::RenderPassTimestampWrites {
549                        query_set: &timer.query_set,
550                        beginning_of_pass_write_index: Some(0),
551                        end_of_pass_write_index: Some(1),
552                    }
553                }),
554                occlusion_query_set: None,
555                multiview_mask: None,
556            });
557            pass.set_pipeline(&self.pipeline);
558            pass.set_bind_group(0, &self.bind_group, &[]);
559            pass.set_vertex_buffer(0, self.index_buffer.slice(..));
560            pass.set_viewport(viewport[0], viewport[1], viewport[2], viewport[3], 0.0, 1.0);
561            pass.draw(0..6, 0..self.telemetry.visible);
562        }
563        let mut timestamp_readback = None;
564        if let Some(timer) = self.gpu_timer.as_mut() {
565            if let Some((buffer, pending)) = timer.reserve_readback() {
566                encoder.resolve_query_set(&timer.query_set, 0..2, &timer.resolve_buffer, 0);
567                encoder.copy_buffer_to_buffer(
568                    &timer.resolve_buffer,
569                    0,
570                    &buffer,
571                    0,
572                    GPU_TIMESTAMP_BYTES,
573                );
574                timestamp_readback = Some((
575                    buffer,
576                    pending,
577                    timer.latest_ms_bits.clone(),
578                    timer.timestamp_period_ns,
579                ));
580            }
581        }
582        self.queue.submit([encoder.finish()]);
583        if let Some((buffer, pending, latest_ms_bits, timestamp_period_ns)) = timestamp_readback {
584            let callback_buffer = buffer.clone();
585            buffer
586                .slice(..GPU_TIMESTAMP_BYTES)
587                .map_async(wgpu::MapMode::Read, move |result| {
588                    if result.is_ok() {
589                        let mapped_result = callback_buffer
590                            .slice(..GPU_TIMESTAMP_BYTES)
591                            .get_mapped_range();
592                        if let Ok(mapped) = mapped_result {
593                            let begin = u64::from_le_bytes(mapped[0..8].try_into().unwrap());
594                            let end = u64::from_le_bytes(mapped[8..16].try_into().unwrap());
595                            if end >= begin {
596                                let elapsed_ms =
597                                    (end - begin) as f64 * timestamp_period_ns / 1_000_000.0;
598                                latest_ms_bits.store(elapsed_ms.to_bits(), AtomicOrdering::Relaxed);
599                            }
600                            drop(mapped);
601                        }
602                        callback_buffer.unmap();
603                    }
604                    pending.store(false, AtomicOrdering::Relaxed);
605                });
606        }
607        self.queue.present(frame);
608        self.inspection_frame = Some(inspection_frame);
609        Ok(())
610    }

690struct ParsedPly {
691    source: Vec<ResearchSplat>,
692    payload_offset: usize,
693    payload_end: usize,
694}
695
696fn parse_stg_ply(bytes: &[u8]) -> Result<ParsedPly, String> {
697    const END_HEADER: &[u8] = b"end_header\n";
698    let header_start = bytes
699        .windows(END_HEADER.len())
700        .position(|window| window == END_HEADER)
701        .ok_or_else(|| "PLY header is missing end_header".to_string())?;
702    let payload_offset = header_start + END_HEADER.len();
703    let header = std::str::from_utf8(&bytes[..payload_offset])
704        .map_err(|_| "PLY header is not UTF-8".to_string())?;
705    if !header.contains("format binary_little_endian 1.0") {
706        return Err("Expected a binary little-endian PLY".to_string());
707    }
708    let count = header
709        .lines()
710        .find_map(|line| line.strip_prefix("element vertex "))
711        .ok_or_else(|| "PLY vertex count is missing".to_string())?
712        .parse::<usize>()
713        .map_err(|_| "PLY vertex count is invalid".to_string())?;
714    let properties: Vec<_> = header
715        .lines()
716        .filter_map(|line| line.strip_prefix("property float "))
717        .collect();
718    if properties.len() != FLOATS_PER_SPLAT {
719        return Err(format!(
720            "Expected the 32-field STG-Lite layout, found {} fields",
721            properties.len()
722        ));
723    }
724    for (index, (actual, expected)) in properties.iter().zip(STG_PROPERTIES).enumerate() {
725        if *actual != expected {
726            return Err(format!(
727                "STG-Lite field {index} must be {expected}, found {actual}"
728            ));
729        }
730    }
731    if count == 0 || count > MAX_SPLATS {
732        return Err(format!(
733            "Expected 1..={MAX_SPLATS} STG-Lite splats, found {count}"
734        ));
735    }
736    let payload_end = count
737        .checked_mul(BYTES_PER_SPLAT)
738        .and_then(|length| payload_offset.checked_add(length))
739        .ok_or_else(|| "PLY payload length overflowed".to_string())?;
740    if payload_end > bytes.len() {
741        return Err("PLY payload is truncated".to_string());
742    }
743
744    let payload = &bytes[payload_offset..payload_end];
745    let mut source = Vec::with_capacity(count);
746    for (index, record) in payload.chunks_exact(BYTES_PER_SPLAT).enumerate() {
747        let value = |index: usize| {
748            let offset = index * 4;
749            f32::from_le_bytes(record[offset..offset + 4].try_into().unwrap())
750        };
751        if (0..FLOATS_PER_SPLAT).any(|field| !value(field).is_finite()) {
752            return Err(format!(
753                "STG-Lite record {index} contains a non-finite value"
754            ));
755        }
756        if [4, 21, 22, 23]
757            .iter()
758            .any(|&field| !value(field).exp().is_finite())
759        {
760            return Err(format!(
761                "STG-Lite record {index} has an overflowing log scale"
762            ));
763        }
764        source.push(ResearchSplat {
765            base: Vec3::new(value(0), value(1), value(2)),
766            temporal_center: value(3),
767            temporal_scale: value(4).exp().max(1e-6),
768            motion_linear: Vec3::new(value(8), value(9), value(10)),
769            motion_quadratic: Vec3::new(value(11), value(12), value(13)),
770            motion_cubic: Vec3::new(value(14), value(15), value(16)),
771            opacity_logit: value(20),
772        });
773    }
774    Ok(ParsedPly {
775        source,
776        payload_offset,
777        payload_end,
778    })
779}

781fn sigmoid(value: f32) -> f32 {
782    1.0 / (1.0 + (-value).exp())
783}
