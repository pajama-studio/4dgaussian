1//! Surface-independent STG-Lite pass. The host owns the device, attachments,
2//! camera, playback clock, submission and presentation. Time is normalized;
3//! this API never guesses the capture duration from a PLY file.
4use crate::{parse_stg_ply, CameraUniform, ResearchSplat};
5use glam::Mat4;
6use wgpu::util::DeviceExt;
7
8#[derive(Clone, Copy, Debug)]
9pub struct PreparedFrame {
10    pub source_count: usize,
11    pub visible_count: usize,
12    pub index_upload_bytes: usize,
13}

15pub struct StgPass {
16    pipeline: wgpu::RenderPipeline,
17    bind_group: wgpu::BindGroup,
18    camera_buffer: wgpu::Buffer,
19    index_buffer: wgpu::Buffer,
20    source: Vec<ResearchSplat>,
21    ordered: Vec<(f32, u32)>,
22    indices: Vec<u32>,
23}
24
25impl StgPass {
26    /// `depth` must match the host attachment. Gaussian depth writes are always
27    /// disabled. Choose LessEqual for conventional Z or GreaterEqual for reverse Z.
28    /// Colors retain the baseline's learned RGB convention; the host must decide
29    /// how to convert that convention before compositing into an HDR scene.
30    pub fn new(
31        device: &wgpu::Device,
32        ply: &[u8],
33        color_format: wgpu::TextureFormat,
34        depth: Option<(wgpu::TextureFormat, wgpu::CompareFunction)>,
35    ) -> Result<Self, String> {
36        let parsed = parse_stg_ply(ply)?;
37        let payload = &ply[parsed.payload_offset..parsed.payload_end];
38        if payload.len() as u64 > u64::from(device.limits().max_storage_buffer_binding_size) {
39            return Err("STG source exceeds the device storage binding limit".into());
40        }
41        let camera_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
42            label: Some("stg-host-camera"),
43            contents: bytemuck::bytes_of(&<CameraUniform as bytemuck::Zeroable>::zeroed()),
44            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
45        });
46        let source_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
47            label: Some("stg-host-source"),
48            contents: payload,
49            usage: wgpu::BufferUsages::STORAGE,
50        });
51        let count = parsed.source.len();
52        let index_buffer = device.create_buffer(&wgpu::BufferDescriptor {
53            label: Some("stg-host-order"),
54            size: (count * 4) as u64,
55            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
56            mapped_at_creation: false,
57        });
58        let layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
59            label: Some("stg-host-layout"),
60            entries: &[
61                wgpu::BindGroupLayoutEntry {
62                    binding: 0,
63                    visibility: wgpu::ShaderStages::VERTEX,
64                    ty: wgpu::BindingType::Buffer {
65                        ty: wgpu::BufferBindingType::Uniform,
66                        has_dynamic_offset: false,
67                        min_binding_size: None,
68                    },
69                    count: None,
70                },
71                wgpu::BindGroupLayoutEntry {
72                    binding: 1,
73                    visibility: wgpu::ShaderStages::VERTEX,
74                    ty: wgpu::BindingType::Buffer {
75                        ty: wgpu::BufferBindingType::Storage { read_only: true },
76                        has_dynamic_offset: false,
77                        min_binding_size: None,
78                    },
79                    count: None,
80                },
81            ],
82        });
83        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
84            label: Some("stg-host-bindings"),
85            layout: &layout,
86            entries: &[
87                wgpu::BindGroupEntry {
88                    binding: 0,
89                    resource: camera_buffer.as_entire_binding(),
90                },
91                wgpu::BindGroupEntry {
92                    binding: 1,
93                    resource: source_buffer.as_entire_binding(),
94                },
95            ],
96        });
97        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
98            label: Some("stg-shared-wgsl"),
99            source: wgpu::ShaderSource::Wgsl(include_str!("splat.wgsl").into()),
100        });
101        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
102            label: Some("stg-host-pipeline-layout"),
103            bind_group_layouts: &[Some(&layout)],
104            immediate_size: 0,
105        });
106        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
107            label: Some("stg-host-pipeline"),
108            layout: Some(&pipeline_layout),
109            vertex: wgpu::VertexState {
110                module: &shader,
111                entry_point: Some("vs_main"),
112                compilation_options: Default::default(),
113                buffers: &[Some(wgpu::VertexBufferLayout {
114                    array_stride: 4,
115                    step_mode: wgpu::VertexStepMode::Instance,
116                    attributes: &[wgpu::VertexAttribute {
117                        format: wgpu::VertexFormat::Uint32,
118                        offset: 0,
119                        shader_location: 0,
120                    }],
121                })],
122            },
123            fragment: Some(wgpu::FragmentState {
124                module: &shader,
125                entry_point: Some("fs_main"),
126                compilation_options: Default::default(),
127                targets: &[Some(wgpu::ColorTargetState {
128                    format: color_format,
129                    blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
130                    write_mask: wgpu::ColorWrites::ALL,
131                })],
132            }),
133            primitive: wgpu::PrimitiveState::default(),
134            depth_stencil: depth.map(|(format, depth_compare)| wgpu::DepthStencilState {
135                format,
136                depth_write_enabled: Some(false),
137                depth_compare: Some(depth_compare),
138                stencil: Default::default(),
139                bias: Default::default(),
140            }),
141            multisample: Default::default(),
142            multiview_mask: None,
143            cache: None,
144        });
145        Ok(Self {
146            pipeline,
147            bind_group,
148            camera_buffer,
149            index_buffer,
150            source: parsed.source,
151            ordered: Vec::with_capacity(count),
152            indices: Vec::with_capacity(count),
153        })
154    }
155
156    /// Matrices are column-major (WebGPU clip Z in 0..1, RH view forward -Z).
157    /// Set the render pass viewport to the same pixel dimensions before draw.
158    /// Submit each prepared frame before preparing another using these buffers.
159    pub fn prepare(
160        &mut self,
161        queue: &wgpu::Queue,
162        normalized_time: f32,
163        view_columns: [[f32; 4]; 4],
164        projection_columns: [[f32; 4]; 4],
165        viewport: [u32; 2],
166    ) -> Result<PreparedFrame, String> {
167        if !normalized_time.is_finite()
168            || !(0.0..=1.0).contains(&normalized_time)
169            || viewport.contains(&0)
170            || view_columns
171                .iter()
172                .chain(&projection_columns)
173                .flatten()
174                .any(|x| !x.is_finite())
175        {
176            return Err("Invalid time, viewport or camera matrix".into());
177        }
178        let view = Mat4::from_cols_array_2d(&view_columns);
179        let view_projection = Mat4::from_cols_array_2d(&projection_columns) * view;
180        self.ordered.clear();
181        for (index, source) in self.source.iter().enumerate() {
182            let (center, opacity) = source.sample(normalized_time);
183            if opacity < 1.0 / 255.0 {
184                continue;
185            }
186            let clip = view_projection * center.extend(1.0);
187            if clip.w <= 0.0 {
188                continue;
189            }
190            let ndc = clip.truncate() / clip.w;
191            // Matches the browser baseline's center-margin heuristic. This is
192            // not a mathematically conservative bound for arbitrarily big splats.
193            if ndc.x.abs() > 1.35 || ndc.y.abs() > 1.35 || !(0.0..=1.0).contains(&ndc.z) {
194                continue;
195            }
196            self.ordered
197                .push((-(view * center.extend(1.0)).z, index as u32));
198        }
199        self.ordered
200            .sort_unstable_by(|a, b| b.0.total_cmp(&a.0).then(a.1.cmp(&b.1)));
201        self.indices.clear();
202        self.indices
203            .extend(self.ordered.iter().map(|entry| entry.1));
204        let [width, height] = viewport.map(|v| v as f32);
205        let camera = CameraUniform {
206            view_proj: view_projection.to_cols_array_2d(),
207            viewport: [width, height, width.recip(), height.recip()],
208            scene: [normalized_time, 0.0, 0.0, 0.0],
209        };
210        queue.write_buffer(&self.camera_buffer, 0, bytemuck::bytes_of(&camera));
211        if !self.indices.is_empty() {
212            queue.write_buffer(&self.index_buffer, 0, bytemuck::cast_slice(&self.indices));
213        }
214        Ok(PreparedFrame {
215            source_count: self.source.len(),
216            visible_count: self.indices.len(),
217            index_upload_bytes: self.indices.len() * 4,
218        })
219    }
220
221    pub fn draw(&self, pass: &mut wgpu::RenderPass<'_>) {
222        pass.set_pipeline(&self.pipeline);
223        pass.set_bind_group(0, &self.bind_group, &[]);
224        pass.set_vertex_buffer(0, self.index_buffer.slice(..));
225        pass.draw(0..6, 0..self.indices.len() as u32);
226    }
227}
