1//! Surface-independent STG-Lite pass. The host owns the device, attachments,
2//! camera, playback clock, submission and presentation. Time is normalized;
3//! this API never guesses the capture duration from a PLY file.
4use crate::stg_prepare::StgPreparation;
5use crate::{parse_stg_ply, CameraUniform};
6use glam::Mat4;
7use wgpu::util::DeviceExt;

9#[derive(Clone, Copy, Debug)]
10pub struct PreparedFrame {
11    pub source_count: usize,
12    pub visible_count: usize,
13    pub index_upload_bytes: usize,
14}

16pub struct StgPass {
17    pipeline: wgpu::RenderPipeline,
18    bind_group: wgpu::BindGroup,
19    camera_buffer: wgpu::Buffer,
20    index_buffer: wgpu::Buffer,
21    preparation: StgPreparation,
22    cpu_variant: u8,

24}

26impl StgPass {
27    /// `depth` must match the host attachment. Gaussian depth writes are always
28    /// disabled. Choose LessEqual for conventional Z or GreaterEqual for reverse Z.
29    /// Colors retain the baseline's learned RGB convention; the host must decide
30    /// how to convert that convention before compositing into an HDR scene.
31    pub fn new(
32        device: &wgpu::Device,
33        ply: &[u8],
34        color_format: wgpu::TextureFormat,
35        depth: Option<(wgpu::TextureFormat, wgpu::CompareFunction)>,
36    ) -> Result<Self, String> {
37        Self::with_options(device, ply, color_format, depth, 9, 0)
38    }

40    /// Benchmark options retain the unmodified baseline for paired comparisons.
41    pub fn with_options(
42        device: &wgpu::Device,
43        ply: &[u8],
44        color_format: wgpu::TextureFormat,
45        depth: Option<(wgpu::TextureFormat, wgpu::CompareFunction)>,
46        cpu_variant: u8,
47        gpu_variant: u8,
48    ) -> Result<Self, String> {
49        let parsed = parse_stg_ply(ply)?;
50        let mut payload = ply[parsed.payload_offset..parsed.payload_end].to_vec();
51        if gpu_variant >= 1 {
52            for row in payload.chunks_exact_mut(128) {
53                for field in [4, 20, 21, 22, 23] {
54                    let v = f32::from_le_bytes(row[field * 4..field * 4 + 4].try_into().unwrap());
55                    let activated = if field == 20 {
56                        crate::sigmoid(v)
57                    } else if field == 4 {
58                        v.exp().max(1e-6)
59                    } else {
60                        v.exp()
61                    };
62                    row[field * 4..field * 4 + 4].copy_from_slice(&activated.to_le_bytes());
63                }
64            }
65        }
66        if payload.len() as u64 > u64::from(device.limits().max_storage_buffer_binding_size) {
67            return Err("STG source exceeds the device storage binding limit".into());
68        }
69        let camera_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
70            label: Some("stg-host-camera"),
71            contents: bytemuck::bytes_of(&<CameraUniform as bytemuck::Zeroable>::zeroed()),
72            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
73        });
74        let source_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
75            label: Some("stg-host-source"),
76            contents: &payload,
77            usage: wgpu::BufferUsages::STORAGE,
78        });
79        let count = parsed.source.len();
80        let index_buffer = device.create_buffer(&wgpu::BufferDescriptor {
81            label: Some("stg-host-order"),
82            size: (count * 4) as u64,
83            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
84            mapped_at_creation: false,
85        });
86        let layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
87            label: Some("stg-host-layout"),
88            entries: &[
89                wgpu::BindGroupLayoutEntry {
90                    binding: 0,
91                    visibility: wgpu::ShaderStages::VERTEX,
92                    ty: wgpu::BindingType::Buffer {
93                        ty: wgpu::BufferBindingType::Uniform,
94                        has_dynamic_offset: false,
95                        min_binding_size: None,
96                    },
97                    count: None,
98                },
99                wgpu::BindGroupLayoutEntry {
100                    binding: 1,
101                    visibility: wgpu::ShaderStages::VERTEX,
102                    ty: wgpu::BindingType::Buffer {
103                        ty: wgpu::BufferBindingType::Storage { read_only: true },
104                        has_dynamic_offset: false,
105                        min_binding_size: None,
106                    },
107                    count: None,
108                },
109            ],
110        });
111        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
112            label: Some("stg-host-bindings"),
113            layout: &layout,
114            entries: &[
115                wgpu::BindGroupEntry {
116                    binding: 0,
117                    resource: camera_buffer.as_entire_binding(),
118                },
119                wgpu::BindGroupEntry {
120                    binding: 1,
121                    resource: source_buffer.as_entire_binding(),
122                },
123            ],
124        });
125        let mut shader_code = include_str!("splat.wgsl").to_string();
126        if gpu_variant >= 1 {
127            shader_code = shader_code
128                .replace("max(exp(splat.r1.x), 0.000001)", "splat.r1.x")
129                .replace("(1.0 / (1.0 + exp(-splat.r5.x)))", "splat.r5.x")
130                .replace("exp(splat.r5.yzw)", "splat.r5.yzw");
131        }
132        if gpu_variant >= 2 {
133            let begin = shader_code.find("var corners =").unwrap();
134            let end = shader_code[begin..].find("return corners[index];").unwrap() + begin;
135            shader_code.replace_range(begin..end, "var corners = array<vec2<f32>, 4>(vec2<f32>(-1.0,-1.0),vec2<f32>(1.0,-1.0),vec2<f32>(-1.0,1.0),vec2<f32>(1.0,1.0));\n  ");
136        }
137        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
138            label: Some("stg-shared-wgsl"),
139            source: wgpu::ShaderSource::Wgsl(shader_code.into()),
140        });
141        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
142            label: Some("stg-host-pipeline-layout"),
143            bind_group_layouts: &[Some(&layout)],
144            immediate_size: 0,
145        });
146        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
147            label: Some("stg-host-pipeline"),
148            layout: Some(&pipeline_layout),
149            vertex: wgpu::VertexState {
150                module: &shader,
151                entry_point: Some("vs_main"),
152                compilation_options: Default::default(),
153                buffers: &[Some(wgpu::VertexBufferLayout {
154                    array_stride: 4,
155                    step_mode: wgpu::VertexStepMode::Instance,
156                    attributes: &[wgpu::VertexAttribute {
157                        format: wgpu::VertexFormat::Uint32,
158                        offset: 0,
159                        shader_location: 0,
160                    }],
161                })],
162            },
163            fragment: Some(wgpu::FragmentState {
164                module: &shader,
165                entry_point: Some("fs_main"),
166                compilation_options: Default::default(),
167                targets: &[Some(wgpu::ColorTargetState {
168                    format: color_format,
169                    blend: Some(wgpu::BlendState::PREMULTIPLIED_ALPHA_BLENDING),
170                    write_mask: wgpu::ColorWrites::ALL,
171                })],
172            }),
173            primitive: wgpu::PrimitiveState {
174                topology: if gpu_variant >= 2 {
175                    wgpu::PrimitiveTopology::TriangleStrip
176                } else {
177                    wgpu::PrimitiveTopology::TriangleList
178                },
179                ..Default::default()
180            },
181            depth_stencil: depth.map(|(format, depth_compare)| wgpu::DepthStencilState {
182                format,
183                depth_write_enabled: Some(false),
184                depth_compare: Some(depth_compare),
185                stencil: Default::default(),
186                bias: Default::default(),
187            }),
188            multisample: Default::default(),
189            multiview_mask: None,
190            cache: None,
191        });
192        Ok(Self {
193            pipeline,
194            bind_group,
195            camera_buffer,
196            index_buffer,
197            preparation: StgPreparation::new(parsed.source),
198            cpu_variant,
199            vertices: if gpu_variant >= 2 { 4 } else { 6 },
200        })
201    }
202
203    /// Matrices are column-major (WebGPU clip Z in 0..1, RH view forward -Z).
204    /// Set the render pass viewport to the same pixel dimensions before draw.
205    /// Submit each prepared frame before preparing another using these buffers.
206    pub fn prepare(
207        &mut self,
208        queue: &wgpu::Queue,
209        normalized_time: f32,
210        view_columns: [[f32; 4]; 4],
211        projection_columns: [[f32; 4]; 4],
212        viewport: [u32; 2],
213    ) -> Result<PreparedFrame, String> {
214        if !normalized_time.is_finite()
215            || !(0.0..=1.0).contains(&normalized_time)
216            || viewport.contains(&0)
217            || view_columns
218                .iter()
219                .chain(&projection_columns)
220                .flatten()
221                .any(|x| !x.is_finite())
222        {
223            return Err("Invalid time, viewport or camera matrix".into());
224        }
225        let view = Mat4::from_cols_array_2d(&view_columns);
226        let projection = Mat4::from_cols_array_2d(&projection_columns);
227        let view_projection = projection * view;
228        let [width, height] = viewport.map(|v| v as f32);
229        self.preparation
230            .prepare(normalized_time, view, projection, self.cpu_variant);
231        let camera = CameraUniform {
232            view_proj: view_projection.to_cols_array_2d(),
233            viewport: [width, height, width.recip(), height.recip()],
234            scene: [normalized_time, 0.0, 0.0, 0.0],
235        };
236        queue.write_buffer(&self.camera_buffer, 0, bytemuck::bytes_of(&camera));
237        if !self.preparation.indices().is_empty() && !self.preparation.cache_hit {
238            queue.write_buffer(
239                &self.index_buffer,
240                0,
241                bytemuck::cast_slice(self.preparation.indices()),
242            );
243        }
244        Ok(PreparedFrame {
245            source_count: self.preparation.len(),
246            visible_count: self.preparation.indices().len(),
247            index_upload_bytes: if self.preparation.cache_hit {
248                0
249            } else {
250                self.preparation.indices().len() * 4
251            },
252        })
253    }
254
255    pub fn draw(&self, pass: &mut wgpu::RenderPass<'_>) {
256        pass.set_pipeline(&self.pipeline);
257        pass.set_bind_group(0, &self.bind_group, &[]);
258        pass.set_vertex_buffer(0, self.index_buffer.slice(..));
259        pass.draw(0..self.vertices, 0..self.preparation.indices().len() as u32);
260    }
261}
