1//! Reproducible CPU preparation variants. Variant 0 preserves the native baseline.
2//! Source IDs and the 1/255 visibility threshold never change between variants.
3use crate::{parse_stg_ply, sigmoid, ResearchSplat};
4use glam::{Mat4, Vec3};

6const CUT: f32 = 1.0 / 255.0;
7const BUCKETS: usize = 64;
8

38        self.buckets.iter().map(|b| b.len() * 4).sum()

40    fn build(source: Vec<ResearchSplat>, force_index: bool) -> Self {
41        let opacity: Vec<_> = source.iter().map(|s| sigmoid(s.opacity_logit)).collect();
42        // Expand the analytic support slightly. The exact baseline test is still
43        // evaluated afterwards; this index must only exclude definitely invisible IDs.
44        let radius: Vec<_> = source
45            .iter()
46            .zip(&opacity)
47            .map(|(s, &a)| {
48                if a < CUT {
49                    -1.0
50                } else {
51                    s.temporal_scale * (a / (CUT * (1.0 - 1e-5))).ln().max(0.0).sqrt() + 1e-5
52                }
53            })
54            .collect();
55        let inner: Vec<_> = source
56            .iter()
57            .zip(&opacity)
58            .map(|(s, &a)| {
59                if a <= CUT * (1.0 + 1e-5) {
60                    -1.0
61                } else {
62                    (s.temporal_scale * (a / (CUT * (1.0 + 1e-5))).ln().sqrt() - 1e-5).max(-1.0)
63                }
64            })
65            .collect();
66        let coverage: f64 = radius
67            .iter()
68            .zip(&source)
69            .filter(|(r, _)| **r >= 0.0)
70            .map(|(&r, s)| {
71                ((s.temporal_center + r).min(1.0) - (s.temporal_center - r).max(0.0)).max(0.0)
72                    as f64
73            })
74            .sum::<f64>()
75            / source.len().max(1) as f64;
76        let indexed = force_index || coverage < 0.65;
77        let mut buckets = if indexed {
78            vec![Vec::new(); BUCKETS]
79        } else {
80            Vec::new()
81        };
82        for (i, (s, &r)) in source.iter().zip(&radius).enumerate() {
83            if r < 0.0 {
84                continue;
85            }
86            for (b, ids) in buckets.iter_mut().enumerate() {
87                if s.temporal_center + r >= b as f32 / BUCKETS as f32
88                    && s.temporal_center - r <= (b + 1) as f32 / BUCKETS as f32
89                {
90                    ids.push(i as u32);
91                }
92            }
93        }
94        let n = source.len();
95        Self {
96            source,
97            opacity,
98            radius,
99            inner,
100            buckets,
101            ordered: Vec::with_capacity(n),
102            scratch: Vec::with_capacity(n),
103            indices: Vec::with_capacity(n),
104            sampled: Vec::with_capacity(n),
105            last_sample: None,
106            last_frame: None,
107            candidates: 0,
108            cache_hit: false,
109            sort_ms: 0.0,
110        }
111    }

126    /// 0 baseline; 1 cached sigmoid; 2 temporal support rejection;
127    /// 3 temporal buckets; 4 exact radix order; 5 repeated-frame cache;
128    /// 6 reuse temporal samples when only the camera changes.
129    pub fn prepare(&mut self, time: f32, view: Mat4, projection: Mat4, variant: u8) -> &[u32] {
130        let key = (time, view, projection, variant);
131        self.cache_hit = variant >= 5 && self.last_frame == Some(key);
132        if self.cache_hit {
133            self.sort_ms = 0.0;
134            return &self.indices;
135        }
136        let vp = projection * view;
137        self.ordered.clear();
138        let bucket = ((time.clamp(0.0, 1.0) * BUCKETS as f32) as usize).min(BUCKETS - 1);
139        let capture_samples = variant >= 6 && (variant < 9 || self.last_sample == Some(time));
140        let indexed = variant >= 3 && !self.buckets.is_empty();
141        let reuse = variant >= 6 && self.last_sample == Some(time) && !self.sampled.is_empty();
142        if !reuse {
143            self.sampled.clear();
144            self.candidates = if indexed {
145                self.buckets[bucket].len()
146            } else {
147                self.source.len()
148            };
149            for j in 0..self.candidates {
150                let i = if indexed {
151                    self.buckets[bucket][j] as usize
152                } else {
153                    j
154                };
155                let s = self.source[i];
156                let dt = time - s.temporal_center;
157                if variant >= 2 && (self.radius[i] < 0.0 || dt.abs() > self.radius[i]) {
158                    continue;
159                }
160                let (center, a) = if variant == 0 {
161                    s.sample(time)
162                } else {
163                    let dt2 = dt * dt;
164                    let center = s.base
165                        + s.motion_linear * dt
166                        + s.motion_quadratic * dt2
167                        + s.motion_cubic * (dt2 * dt);
168                    let a = if variant >= 7 && dt.abs() < self.inner[i] {
169                        1.0
170                    } else {
171                        self.opacity[i] * (-(dt / s.temporal_scale).powi(2)).exp()
172                    };
173                    (center, a)
174                };
175                if a < CUT {
176                    continue;
177                }
178                if capture_samples {
179                    self.sampled.push((center, i as u32));
180                }
181                project(&mut self.ordered, center, i as u32, view, vp);
182            }
183        } else {
184            for &(center, id) in &self.sampled {
185                project(&mut self.ordered, center, id, view, vp);
186            }
187        }
188        self.last_sample = if variant >= 6 { Some(time) } else { None };
189        let sort_started = web_time::Instant::now();
190        if variant >= 4 {
191            if variant >= 8 {
192                radix11(&mut self.ordered, &mut self.scratch);
193            } else {
194                radix(&mut self.ordered, &mut self.scratch);
195            }
196        } else {
197            self.ordered
198                .sort_unstable_by(|a, b| b.0.total_cmp(&a.0).then(a.1.cmp(&b.1)));
199        }
200        self.sort_ms = sort_started.elapsed().as_secs_f64() * 1000.0;
201        self.indices.clear();
202        self.indices.extend(self.ordered.iter().map(|x| x.1));
203        self.last_frame = Some(key);
204        &self.indices
205    }
206}

208fn project(out: &mut Vec<(f32, u32)>, center: Vec3, id: u32, view: Mat4, vp: Mat4) {
209    let clip = vp * center.extend(1.0);
210    if clip.w <= 0.0 {
211        return;
212    }
213    let ndc = clip.truncate() / clip.w;
214    if ndc.x.abs() > 1.35 || ndc.y.abs() > 1.35 || ndc.z < 0.0 || ndc.z > 1.0 {
215        return;
216    }
217    out.push((-(view * center.extend(1.0)).z, id));
218}

220// Stable LSD sort. Input IDs are ascending, so equal depths retain the baseline
221// ID tie-break. Float key also handles signed depths and IEEE total-order values.
222fn radix(values: &mut Vec<(f32, u32)>, temp: &mut Vec<(f32, u32)>) {
223    temp.resize(values.len(), (0.0, 0));
224    let key = |x: f32| {
225        let bits = x.to_bits();
226        !(bits
227            ^ if bits >> 31 != 0 {
228                u32::MAX
229            } else {
230                0x8000_0000
231            })
232    };
233    for shift in [0, 8, 16, 24] {
234        let mut counts = [0usize; 256];
235        for &(depth, _) in values.iter() {
236            counts[((key(depth) >> shift) & 255) as usize] += 1;
237        }
238        let mut sum = 0;
239        for count in &mut counts {
240            let n = *count;
241            *count = sum;
242            sum += n;
243        }
244        for &value in values.iter() {
245            let bucket = ((key(value.0) >> shift) & 255) as usize;
246            temp[counts[bucket]] = value;
247            counts[bucket] += 1;
248        }
249        std::mem::swap(values, temp);
250    }
251}

276fn radix11(values: &mut Vec<(f32, u32)>, temp: &mut Vec<(f32, u32)>) {
277    temp.resize(values.len(), (0.0, 0));
278    let key = |x: f32| {
279        let b = x.to_bits();
280        !(b ^ if b >> 31 != 0 { u32::MAX } else { 0x8000_0000 })
281    };
282    for shift in [0, 11, 22] {
283        let mut counts = [0usize; 2048];
284        for &(depth, _) in values.iter() {
285            counts[((key(depth) >> shift) & 2047) as usize] += 1;
286        }
287        let mut sum = 0;
288        for c in &mut counts {
289            let n = *c;
290            *c = sum;
291            sum += n;
292        }
293        for &v in values.iter() {
294            let b = ((key(v.0) >> shift) & 2047) as usize;
295            temp[counts[b]] = v;
296            counts[b] += 1;
297        }
298        std::mem::swap(values, temp);
299    }
